    const MAX_CONCURRENT_FETCHES = 2;
    const MIN_FETCH_INTERVAL = 220;

    function throttledFetch(input, init) {
        return new Promise((resolve, reject) => {
            fetchQueue.push({ input, init, resolve, reject });
            runFetchQueue();
        });
    }

    function runFetchQueue() {
        if (activeFetches >= MAX_CONCURRENT_FETCHES) return;
        if (fetchQueue.length === 0) return;

        const now = Date.now();
        const waitMs = Math.max(0, MIN_FETCH_INTERVAL - (now - lastFetchTime));
        if (waitMs > 0) {
            setTimeout(runFetchQueue, waitMs);
            return;
        }

        const task = fetchQueue.shift();
        activeFetches += 1;
        lastFetchTime = Date.now();

        fetch(task.input, task.init)
            .then(task.resolve)
            .catch(task.reject)
            .finally(() => {
                activeFetches -= 1;
                runFetchQueue();
            });
    }

        refreshDuration = duration;
        if (countdownInterval) clearInterval(countdownInterval);

        const el  = document.getElementById('refresh-countdown');
        let remaining = duration;
        countdownInterval = setInterval(() => {
            const pct = (remaining / duration) * 100;
            if (bar) bar.style.width = pct + '%';
            if (el) el.innerText = remaining > 0 ? `${Math.ceil(remaining / 1000)}s 後更新` : '更新中...';
            remaining = Math.max(0, remaining - 1000);
        }, 1000);
    }

    function getCityByMapCenter(lat, lng) {
        let minD = Infinity, nearest = 'Taipei';
        for (const [code, center] of Object.entries(cityCenters)) {
            const d = Math.sqrt(Math.pow(lat - center.lat, 2) + Math.pow(lng - center.lng, 2));
            if (d < minD) { minD = d; nearest = code; }
        }
        return minD > 0.5 ? null : nearest;
    }

        if (!selectedValue) return;
        
        let cat = 'CityBus';
        let cityKey = selectedValue;

        // 解析 "分類:縣市" 格式
        if (selectedValue.includes(':')) {
            const parts = selectedValue.split(':');
            cat = parts[0];
            cityKey = parts[1];
        } else {

            cat = cityKey === 'InterCity' ? 'InterCity' : 'CityBus';

        browser.type = cat;

        // 對齊 select value

            updateMarqueeSection('marquee-city-text', browser.city, browser.type);

        container.innerHTML = `<div class="route-cards-grid">${Array(16).fill('<div class="browser-skeleton"></div>').join('')}</div>`;

        try {
            // 🌟 這裡把 category 也傳給 Worker
            const res = await throttledFetch(`${WORKER_URL}/?action=list_all&city=${cityKey}&category=${cat}`);
            const data = await res.json();

                container.innerHTML = `<div class="browser-empty">此區域暫無路線</div>`;
                return;

                const rName = r.name || '';
                const safeName = rName.replace(/'/g, '&#39;');
                const routeSubText = `${r.departure || ''} → ${r.destination || ''}`;
                return `<button class="route-card-btn" onclick="loadRoute('${safeName}', '${r.city}', '${r.type}')">
                    <div class="route-card-num">${rName}</div>
                    <div class="route-card-dest">${routeSubText}</div>
                </button>`;
            }).join('');
            container.innerHTML = `<div class="browser-count-bar">共 <strong>${routes.length}</strong> 條路線</div><div class="route-cards-grid">${cards}</div>`;
        } catch (e) {
            container.innerHTML = `<div class="browser-empty">載入失敗</div>`;
        }
    }

    // 🌟 在這裡接收正確的 category，建立專屬的獨立快取防止污染
    async function loadVehicleData(city, category) {
        const cacheKey = `${category}_${city}`;
        if (vehicleMap[cacheKey]) return;
        vehicleMap[cacheKey] = {}; 
        try {
            const res = await throttledFetch(`${WORKER_URL}/?action=vehicle&category=${category}&city=${city}`);
            const data = await res.json();
            
            if (Array.isArray(data)) {
                data.forEach(v => {
                    vehicleMap[cacheKey][v.PlateNumb] = {
                        year: v.ManufactureYear || (v.PurchaseTime ? v.PurchaseTime.substring(0, 4) : '不詳'),
                        isLowFloor: v.IsLowFloor === 1 || v.HasLiftOrRamp === 1,
                        hasWifi: v.HasWifi === 1,
                        isElectric: v.IsElectric === 1 || v.VehicleType === 1,
                        hasLift: v.HasLiftOrRamp === 1
                    };
                });
            } else if (typeof data === 'object' && data !== null) {
                vehicleMap[cacheKey] = data;
            }
        } catch (e) { 
            console.warn(`車籍字典載入異常 (${city})`, e); 
        }
    }

    // 🌟 在這裡將 action=fare 一併獲取，讓 Timeline 可以立刻顯示緩衝區標籤
    async function loadRoute(name, city, type) {
        const searchInput = document.getElementById('search-input');
        if (searchInput) searchInput.value = name;
        
        document.getElementById('home-view').style.display = 'none';
        document.getElementById('route-view').style.display = 'flex';
        document.getElementById('loading').style.display = 'flex';
        if(document.getElementById('marquee-wrapper')) document.getElementById('marquee-wrapper').style.display = 'none';
        
        currentData.route = name;
        currentData.city = city || browser.city;
        currentData.type = type || browser.type;
        currentData.fare = null;

        // 🌟 正確帶入類別（type）以支援跨種類公車車籍載入
        loadVehicleData(currentData.city, currentData.type);

        try {
            const [infoRes, shapeRes, fareRes] = await Promise.all([
                throttledFetch(`${WORKER_URL}/?action=info&category=${currentData.type}&city=${currentData.city}&route=${encodeURIComponent(name)}`).then(r => r.json()),
                throttledFetch(`${WORKER_URL}/?action=shape&category=${currentData.type}&city=${currentData.city}&route=${encodeURIComponent(name)}`).then(r => r.json()),
                throttledFetch(`${WORKER_URL}/?action=fare&category=${currentData.type}&city=${currentData.city}&route=${encodeURIComponent(name)}`).then(r => r.json()).catch(() => null)
            ]);

            const infoList = Array.isArray(infoRes) ? infoRes : [];
            const dir0 = infoList.find(d => d.Direction == 0) || infoList[0];
            const dir1 = infoList.find(d => d.Direction == 1) || (infoList.length > 1 && infoList[1] !== dir0 ? infoList[1] : null);
            
            currentData.stops = { 0: dir0 ? dir0.Stops : [], 1: dir1 ? dir1.Stops : [] };
            currentData.info = { 0: dir0, 1: dir1 };
            currentData.fare = fareRes ? fareRes.fares : null;

            document.getElementById('route-title').innerText = name;
            
            const getStop = (dirObj, isDep) => {
                if (!dirObj || !dirObj.Stops || !dirObj.Stops.length) return '';
                const idx = isDep ? 0 : dirObj.Stops.length - 1;
                return dirObj.Stops[idx].StopName?.Zh_tw || dirObj.Stops[idx].StopName || '';
            };
            
            const dep0 = getStop(dir0, true);
            const dest0 = getStop(dir0, false);
            const subtitleEl = document.getElementById('route-subtitle');
            
            if (dep0 && dest0) subtitleEl.innerHTML = `${dep0} <i class="fas fa-arrow-right" style="font-size:0.8em; margin:0 4px;"></i> ${dest0}`;
            else subtitleEl.innerText = dest0 || dep0 || '路線詳情';

            document.getElementById('dest-0').innerText = dest0 || '終點';
            document.getElementById('dest-1').innerText = getStop(dir1, false) || '終點';
            
            const tab1 = document.getElementById('tab-1');
            if(tab1) tab1.style.display = dir1 ? 'block' : 'none';
            
            currentData.shapes = Array.isArray(shapeRes) ? shapeRes : [];

            switchDirection(dir0 ? 0 : 1);
        } catch (e) {
            console.error(e);
            alert('載入路線失敗，請重試');
        } finally {
            document.getElementById('loading').style.display = 'none';
        }
    }

    // 🌟 解析新的段次計費格式，取出緩衝區
    function renderFareInfo(dir) {
        try {
            let bStart = null, bEnd = null;
            if (currentData.fare && currentData.fare.length > 0) {
                const dirLabel = dir === 1 ? '返程' : (dir === 0 ? '去程' : '迴圈');
                const fareProfile = currentData.fare.find(f => f.section_fares && f.section_fares.length > 0);
                if (fareProfile) {
                    const section = fareProfile.section_fares.find(s => s.direction === dirLabel) || fareProfile.section_fares[0];
                    if (section && section.buffer_zones && section.buffer_zones.length > 0) {
                        bStart = section.buffer_zones[0].origin;
                        bEnd = section.buffer_zones[0].destination;
                    }
                }
            }
            return { start: bStart, end: bEnd };
        } catch (e) { return { start: null, end: null }; }
    }

    function switchDirection(dir) {
        currentData.direction = dir;
        document.querySelectorAll('.tab').forEach(t => t.classList.remove('active'));
        const tabDir = document.getElementById(`tab-${dir}`);
        if(tabDir) tabDir.classList.add('active');

        const stops = currentData.stops[dir] || [];
        const container = document.getElementById('stops-container');
        
        if (!stops.length) {
            container.innerHTML = '<div class="empty-view">此方向無營運資料</div>';
            drawStaticMap(dir); 
            return;
        }

        let bufferStops = renderFareInfo(dir);

        container.innerHTML = stops.map((s, idx) => {
            const stopName = s.StopName?.Zh_tw || s.StopName || '未命名站牌';
            const isBufferStart = bufferStops.start === stopName;
            const isBufferEnd = bufferStops.end === stopName;
            const bufferHtml = isBufferStart ? '<span class="buffer-tag">緩衝區起點</span>' : (isBufferEnd ? '<span class="buffer-tag">緩衝區終點</span>' : '');
            const uid = s.StopUID || `stop-${idx}`;
            
            return `
                <div class="stop-row ${isBufferStart ? 'buffer-start' : ''} ${isBufferEnd ? 'buffer-end' : ''}" id="stop-${uid}" onclick="focusMap(${s.StopPosition?.PositionLat}, ${s.StopPosition?.PositionLon})">
                    <div class="col-time"><span class="badge-time none">--</span></div>
                    <div class="col-line"><div class="line-bar"></div><div class="line-dot">${idx + 1}</div></div>
                    <div class="col-info">
                        <div class="stop-name">${stopName}${bufferHtml}</div>
                        <div id="bus-${uid}"></div>
                    </div>
                </div>
            `;
        }).join('');

        drawStaticMap(dir);
        refreshLiveData();
        if (refreshInterval) clearInterval(refreshInterval);
        refreshInterval = setInterval(refreshLiveData, 10000);
        startCountdown(10000);
    }

    function getDistance(lat1, lon1, lat2, lon2) {
        if (!lat1 || !lon1 || !lat2 || !lon2) return Infinity;
        const dLat = lat1 - lat2;
        const dLon = lon1 - lon2;
        return Math.sqrt(dLat * dLat + dLon * dLon);
    }

    async function refreshLiveData() {
        const icon = document.getElementById('refresh-icon');
        const updateTimeEl = document.getElementById('update-time');
        if(icon) icon.classList.add('fa-spin');
        startCountdown(refreshDuration);

        try {
            const url = `${WORKER_URL}/?category=${currentData.type}&city=${currentData.city}&route=${encodeURIComponent(currentData.route)}`;
            const res = await throttledFetch(url);
            const data = await res.json();
            
            if (updateTimeEl) updateTimeEl.innerText = new Date().toTimeString().substring(0, 5);

            const estimates = data.estimates || {};
            const buses = (data.buses || []).filter(b => b.dir == currentData.direction);
            lastBuses = buses; 
            
            updateStopsUI(estimates, buses);
            updateBusesMap(buses);
        } catch (e) {
            console.error('Update failed', e);
            if (updateTimeEl) updateTimeEl.innerText = '⚠️ 連線錯誤';
        } finally {
            if(icon) setTimeout(() => icon.classList.remove('fa-spin'), 1000);
        }
    }

    function updateStopsUI(estimates, buses) {
        const stops = currentData.stops[currentData.direction];
        if (!stops) return;
        
        let assignedBuses = new Set();

        stops.forEach(stop => {
            const uid = stop.StopUID;
            const est = estimates[`${currentData.direction}_${uid}`];
            
            const timeEl = document.querySelector(`#stop-${uid} .badge-time`);
            const busEl = document.querySelector(`#bus-${uid}`);
            const rowEl = document.getElementById(`stop-${uid}`);
            
            if (!timeEl || !busEl || !rowEl) return;

            timeEl.className = 'badge-time';
            rowEl.classList.remove('coming', 'has-bus');
            busEl.innerHTML = ''; 

            let isActive = false;

            if (!est) {
                timeEl.innerText = '--';
                timeEl.classList.add('none');
            } else if (est.status === 3) {
                timeEl.innerText = '末班已過';
                timeEl.classList.add('none');
            } else if (est.status === 4) {
                timeEl.innerText = '今日停駛';
                timeEl.classList.add('none');
            } else if (est.status === 1) {
                let showTime = false;
                if (est.nextBusTime) {
                    const nextDate = new Date(est.nextBusTime);
                    if (!isNaN(nextDate.getTime())) {
                        const hours = nextDate.getHours().toString().padStart(2, '0');
                        const mins = nextDate.getMinutes().toString().padStart(2, '0');
                        timeEl.innerText = `${hours}:${mins}`; 
                        timeEl.classList.add('badge-time', 'normal'); 
                        timeEl.style.fontSize = '0.9rem'; 
                        showTime = true;
                    }
                }
                if (!showTime) {
                    timeEl.innerText = '未發車';
                    timeEl.classList.add('none');
                }
            } else if (est.status === 2) {
                timeEl.innerText = '不停靠';
                timeEl.classList.add('none');
            } else if (est.sec != null) {
                isActive = true; 
                const min = Math.floor(est.sec / 60);
                if (min < 1) {
                    timeEl.innerText = '進站中';
                    timeEl.classList.add('coming');
                    rowEl.classList.add('coming');
                } else if (min <= 3) {
                    timeEl.innerText = `${min}分`;
                    timeEl.classList.add('soon');
                } else {
                    timeEl.innerText = `${min}分`;
                    timeEl.classList.add('normal');
                }
            }

            let targetBus = null;
            if (est && est.plate && est.plate !== '-1' && !assignedBuses.has(est.plate)) {
                targetBus = buses.find(b => b.plate === est.plate);
            }
            
            if (!targetBus && isActive) {
                targetBus = buses.find(b => {
                    if (!b.lat || assignedBuses.has(b.plate)) return false;
                    return getDistance(b.lat, b.lon, stop.StopPosition?.PositionLat, stop.StopPosition?.PositionLon) < 0.02; 
                });
            }

            if (targetBus) {
                const plate = targetBus.plate || '---';
                assignedBuses.add(plate);

                let operator = '客運公司';
                if (currentData.info && currentData.info[currentData.direction]?.Operators) {
                    const ops = currentData.info[currentData.direction].Operators;
                    if (Array.isArray(ops) && ops.length > 0) {
                        operator = ops[0].OperatorName?.Zh_tw || ops[0].OperatorName || '客運公司';
                    }
                }

                // 🌟 改用獨立複合鍵抓取對應字典
                const cityRegistry = vehicleMap[`${currentData.type}_${currentData.city}`] || {};
                const vReg = cityRegistry[plate] || {};
                const year = vReg.year && vReg.year !== '不詳' ? `${vReg.year}年車` : '年份不詳';
                // 判斷是否為電動公車：後端 TDX 標示為 true，或是車牌符合 FV 結尾 / EA 開頭
                const isElectric = vReg.isElectric || plate.endsWith('FV') || /^E[A-Z]{2}-/.test(plate); 
                const isLowFloor = vReg.isLowFloor || false;
                const hasWifi = vReg.hasWifi || false;
                const hasLift = vReg.hasLift || false;

    // 🌟 時刻表彈窗 (接軌 Worker 新 API 結構)
    async function showTimetableModal() {
        currentModalSession++; // 🌟 發行新的彈窗身分證
        const session = currentModalSession;

        document.getElementById('options-dropdown').style.display = 'none';
        const modal = document.getElementById('modal-overlay');
        const title = document.getElementById('modal-title');
        const body = document.getElementById('modal-body');
        
        title.innerHTML = `<i class="fas fa-clock"></i> <span>${currentData.route} 時刻資訊</span>`;
        body.innerHTML = `<div style="text-align:center; padding:40px;"><i class="fas fa-spinner fa-spin fa-2x" style="color:var(--primary)"></i><br><br>正在載入時刻表...</div>`;
        modal.style.display = 'flex';

        try {
            const url = `${WORKER_URL}/?action=timetable&category=${currentData.type}&city=${currentData.city}&route=${encodeURIComponent(currentData.route)}`;
            const res = await throttledFetch(url);
            const data = await res.json();
            const scheduleData = data && data.timetables ? data.timetables : [];
            if (currentModalSession !== session) return;
            if (!Array.isArray(scheduleData) || scheduleData.length === 0) {
                body.innerHTML = '<div style="padding:20px; text-align:center; color:#666;">查無班次資料</div>';
                return;
            }

            const currentDirLabel = currentData.direction === 1 ? '返程' : (currentData.direction === 0 ? '去程' : '迴圈');
            const safeId = (text) => text.replace(/\s+/g, '-').replace(/[^A-Za-z0-9_-]/g, '');

            // --- 固定時刻表區塊渲染 ---
            const renderScheduleBlock = (s) => {
                const times = Array.isArray(s.departure_times) ? s.departure_times.filter(Boolean) : [];
                if (!times.length) return '';
                const timeChips = times.map(t => `<div class="time-chip">${t.substring(0,5)}</div>`).join('');
                return `
                    <div style="margin-bottom:16px;">
                        <div style="display:flex; align-items:center; gap:8px; font-weight:700; font-size:0.9rem; color:#334155; margin-bottom:10px;">
                            <i class="fas fa-calendar-check" style="color:#0f766e;"></i>
                            <span>${s.service_day || '未知服務日'}</span>
                            ${s.is_low_floor ? '<span style="font-size:0.7rem; background:#ecfdf5; color:#059669; padding:2px 6px; border-radius:999px;">低地板</span>' : ''}
                        </div>
                        <div class="timetable-grid">${timeChips}</div>
                    </div>
                `;
            };

            // --- 班距發車區塊渲染 ---
            const renderFrequencyBlock = (f) => {
                const rangesHtml = (f.ranges || []).map(r => {
                    const headway = (r.min || r.max) ? `${r.min}${r.max && r.max !== r.min ? `–${r.max}` : ''} 分鐘` : '固定班距';
                    return `
                        <div style="display:flex; justify-content:space-between; padding:8px 0; border-bottom:1px dashed #cbd5e1;">
                            <span style="color:#475569; font-family:monospace; font-size:0.95rem; font-weight:bold;">${r.time_range}</span>
                            <span style="color:#2563eb; font-weight:800;">${headway}</span>
                        </div>
                    `;
                }).join('');

                return `
                    <div style="margin-bottom:16px; background:white; border:1px solid #e2e8f0; border-radius:10px; overflow:hidden;">
                        <div style="background:#f8fafc; padding:10px 14px; font-weight:800; color:#1e3a5f; font-size:0.95rem; border-bottom:1px solid #e2e8f0; display:flex; align-items:center; gap:8px;">
                            <i class="fas fa-calendar-day" style="color:#64748b;"></i> ${f.service_day || '未知服務日'}
                        </div>
                        <div style="padding:2px 14px 6px;">
                            ${rangesHtml}
                        </div>
                    </div>
                `;
            };

            const tabs = scheduleData.map((entry, idx) => {
                const label = entry.direction || `方向${idx + 1}`;
                const active = label === currentDirLabel;
                return `<button class="timetable-tab-btn ${active ? 'active' : ''}" onclick="switchTimetableTab('${safeId(label)}', event)">${label}</button>`;
            }).join('');

            const contents = scheduleData.map((entry) => {
                const label = entry.direction || '方向';
                const isActive = label === currentDirLabel;
                const schedules = Array.isArray(entry.schedules) ? entry.schedules : [];
                const frequencies = Array.isArray(entry.frequencies) ? entry.frequencies : [];
                
                const scheduleHtml = schedules.map(renderScheduleBlock).join('');
                const frequencyHtml = frequencies.map(renderFrequencyBlock).join('');

                let sectionHtml = '';
                
                // 如果有「班距發車」資料
                if (frequencyHtml) {
                    sectionHtml += `<div style="font-weight:900; font-size:1.05rem; color:#1e40af; margin-bottom:12px; margin-top:10px; display:flex; align-items:center; gap:8px;"><i class="fas fa-history"></i> 發車班距</div>`;
                    sectionHtml += frequencyHtml;
                }
                
                // 如果有「固定時刻」資料
                if (scheduleHtml) {
                    if (frequencyHtml) sectionHtml += `<div style="height:1px; background:#e2e8f0; margin:20px 0;"></div>`;
                    sectionHtml += `<div style="font-weight:900; font-size:1.05rem; color:#0f766e; margin-bottom:12px; margin-top:10px; display:flex; align-items:center; gap:8px;"><i class="fas fa-list-ol"></i> 固定發車時刻</div>`;
                    sectionHtml += scheduleHtml;
                }
                
                // 兩者都沒有
                if (!sectionHtml) {
                    sectionHtml = `<div style="text-align:center; color:#94a3b8; padding:30px;"><i class="fas fa-bus-slash fa-2x" style="margin-bottom:10px; color:#cbd5e1;"></i><br>目前無排定營運資料</div>`;
                }

                return `<div id="tt-${safeId(label)}" style="display:${isActive ? 'block' : 'none'};">${sectionHtml}</div>`;
            }).join('');

            body.innerHTML = `
                <div class="timetable-tabs">${tabs}</div>
                ${contents}
            `;

            window.switchTimetableTab = function(textId, e) {
                document.querySelectorAll('.timetable-tab-btn').forEach(b => b.classList.remove('active'));
                if (e && e.target) e.target.classList.add('active');
                
                // 找到所有的 tt-* 元素並隱藏，只顯示選中的
                const container = document.getElementById('modal-body');
                const contents = container.querySelectorAll('[id^="tt-"]');
                contents.forEach(el => {
                    el.style.display = el.id === `tt-${textId}` ? 'block' : 'none';
                });
            };
        } catch (e) {
            console.error("Timetable Error:", e);
            body.innerHTML = '<p style="text-align:center; color:red; padding:20px;">資料載入失敗</p>';
        }
    }

        }

        const fares = currentData.fare || [];
        if (!fares.length) {
            body.innerHTML = `<div style="text-align:center; color:#9ca3af; padding:30px;">目前 TDX 未提供此路線票價</div>`;
            return;
        }

        const stops = currentData.stops[currentData.direction] || [];

        if (stops.length === 0) {
            body.innerHTML = `<div style="text-align:center; color:#9ca3af; padding:30px;">無法取得站牌資料進行試算</div>`;
            return;
        }

        // 產生站牌下拉選單

        body.innerHTML = `
            <div style="background:#f8fafc; padding:16px; border-radius:12px; margin-bottom:16px; border:1px solid #e2e8f0;">

                <div style="display:flex; flex-direction:column; gap:10px;">
                    <div style="display:flex; align-items:center; gap:8px;">
                        <div style="width:32px; height:32px; border-radius:50%; background:#dbeafe; color:#2563eb; display:flex; align-items:center; justify-content:center; font-weight:bold; font-size:0.9rem; flex-shrink:0;">起</div>
                        <select id="fare-from" onchange="calcFare()" style="flex:1; padding:10px; border-radius:8px; border:1px solid #cbd5e1; font-size:0.95rem; outline:none; background:white;">
                            ${stopOptions}
                        </select>
                    </div>
                    <div style="display:flex; align-items:center; gap:8px;">
                        <div style="width:32px; height:32px; border-radius:50%; background:#fee2e2; color:#ef4444; display:flex; align-items:center; justify-content:center; font-weight:bold; font-size:0.9rem; flex-shrink:0;">迄</div>
                        <select id="fare-to" onchange="calcFare()" style="flex:1; padding:10px; border-radius:8px; border:1px solid #cbd5e1; font-size:0.95rem; outline:none; background:white;">
                            ${stopOptions}
                        </select>
                    </div>
                </div>
            </div>
            <div id="fare-result-container"></div>
        `;

        // 預設將迄站設為最後一站
        document.getElementById('fare-to').value = stops.length - 1;

        // 計算票價的核心邏輯
        window.calcFare = function() {
            const fromIdx = parseInt(document.getElementById('fare-from').value);
            const toIdx = parseInt(document.getElementById('fare-to').value);
            const resEl = document.getElementById('fare-result-container');

            if (fromIdx >= toIdx) {
                resEl.innerHTML = `<div style="text-align:center; padding:15px; color:#ef4444; font-weight:bold; background:#fef2f2; border-radius:8px;">上車點必須在下車點之前</div>`;
                return;
            }

            if (fareProfile.is_free) {
                 resEl.innerHTML = `<div style="text-align:center; padding:20px; color:#059669; font-size:1.2rem; font-weight:900;">此路線免費搭乘 🆓</div>`;
                 return;
            }

            let resultFares = [];
            let segmentHtml = '';

            if (isOD) {
                // 【客運 / 里程計費】
                const fromName = stops[fromIdx].StopName?.Zh_tw || stops[fromIdx].StopName;
                const toName = stops[toIdx].StopName?.Zh_tw || stops[toIdx].StopName;
                const odFares = fareProfile.od_fares || [];
                
                // 尋找包含關鍵字的區間 (因為TDX客運站名有時候會有中英文夾雜或備註不同)
                const match = odFares.find(od => 
                    (od.origin.includes(fromName) || fromName.includes(od.origin)) && 
                    (od.destination.includes(toName) || toName.includes(od.destination))
                );
                
                if (match) {
                    resultFares = match.fares;
                } else {
                     resEl.innerHTML = `<div style="text-align:center; padding:20px; color:#64748b; background:#f1f5f9; border-radius:8px;">無法精確匹配此區間的里程票價<br><span style="font-size:0.8rem;">(TDX 起訖站名可能不完全對應)</span></div>`;
                     return;
                }
            } else {
                // 【市區公車 / 段次計費】
                const secFares = fareProfile.section_fares && fareProfile.section_fares.length > 0 ? fareProfile.section_fares[0] : null;
                if (!secFares) {
                    resEl.innerHTML = `<div style="text-align:center; padding:20px; color:#64748b;">該路線無段次票價明細設定</div>`;
                    return;
                }

                let segments = 1;
                const buffers = secFares.buffer_zones || [];
                
                // 自動偵測是否跨越緩衝區增加段次
                buffers.forEach(bz => {
                    const bStartIdx = stops.findIndex(s => (s.StopName?.Zh_tw || s.StopName) === bz.origin);
                    const bEndIdx = stops.findIndex(s => (s.StopName?.Zh_tw || s.StopName) === bz.destination);
                    
                    if (bStartIdx !== -1 && bEndIdx !== -1) {
                        const trueStart = Math.min(bStartIdx, bEndIdx);
                        const trueEnd = Math.max(bStartIdx, bEndIdx);
                        // 規則: 若上車於緩衝區前，且下車於緩衝區後，則加收一段票
                        if (fromIdx < trueStart && toIdx > trueEnd) segments++;
                    }
                });

                const baseFares = secFares.fares || [];
                resultFares = baseFares.map(f => ({ ...f, price: f.price * segments }));
                segmentHtml = `<div style="text-align:center; font-weight:800; color:#d97706; margin-bottom:12px; background:#fffbeb; padding:8px; border-radius:8px; border:1px solid #fef3c7;">本區間需收取 ${segments} 段票</div>`;
            }

            if (!resultFares || resultFares.length === 0) {
                 resEl.innerHTML = `<div style="text-align:center; padding:20px; color:#64748b;">無票價資料</div>`;
                 return;
            }

            // 渲染最終金額
            resEl.innerHTML = `
                ${segmentHtml}
                <div class="fare-types-grid">
                    ${resultFares.map(f => `
                        <div class="fare-type-cell ${f.class === '成人' || f.class === '全票' || f.class === '一般票' ? 'highlight' : ''}">
                            <div class="fare-type-label">${f.class || '票種'}</div>
                            <div class="fare-type-price">$${f.price}</div>
                        </div>
                    `).join('')}
                </div>
            `;
        };

        // 第一次開啟自動執行一次計算
        window.calcFare();
    }

    function parseWKT(wkt) {
        if (!wkt) return [];
        const points = [];
        const matches = wkt.match(/-?\d+\.\d+\s+-?\d+\.\d+/g);
        if (matches) {
            matches.forEach(match => {
                const [lng, lat] = match.split(/\s+/);
                points.push({ lat: parseFloat(lat), lng: parseFloat(lng) });
            });
        }
        return points;
    }

    function drawStaticMap(dir) {
        if(!map) return;
        
        mapMarkers.stops.forEach(m => m.setMap(null));
        mapMarkers.stops = [];
        
        if (routeLine) routeLine.setMap(null); 

        let path = [];
        if (currentData.shapes && currentData.shapes.length > 0) {
            const shapeObj = currentData.shapes.find(s => s.Direction == dir) || currentData.shapes[0];
            if (shapeObj && shapeObj.Geometry) {
                path = parseWKT(shapeObj.Geometry);
            }
        }

        if (path.length > 0) {
            routeLine = new google.maps.Polyline({
                path: path,
                geodesic: true,
                strokeColor: '#2563eb', 
                strokeOpacity: 0.8,
                strokeWeight: 5,
                map: map
            });
            
            const bounds = new google.maps.LatLngBounds();
            path.forEach(p => bounds.extend(p));
            map.fitBounds(bounds);
        }

        const stops = currentData.stops[dir] || [];
        stops.forEach((s, idx) => {
            const dotEl = document.createElement('div');
            dotEl.style.width = '8px';
            dotEl.style.height = '8px';
            dotEl.style.backgroundColor = '#2563eb'; 
            dotEl.style.borderRadius = '50%';
            dotEl.style.border = '1px solid white';
            dotEl.style.boxShadow = '0 1px 2px rgba(0,0,0,0.3)';
            
            const pinEl = new google.maps.marker.PinElement({
                glyphText: (idx + 1).toString(),
                glyphColor: 'white',
                background: '#2563eb', 
                borderColor: 'white',
                scale: 0.8
            });

            const marker = new google.maps.marker.AdvancedMarkerElement({
                position: { lat: s.StopPosition.PositionLat, lng: s.StopPosition.PositionLon },
                map: map,
                content: dotEl, 
                title: s.StopName?.Zh_tw || s.StopName,
                zIndex: 10 
            });

            marker.dotContent = dotEl;
            marker.pinContent = pinEl.element;

            marker.addListener('gmp-click', () => {
                if (activeInfoWindow) activeInfoWindow.close();
                const infoWindow = new google.maps.InfoWindow();
                const stopName = s.StopName?.Zh_tw || s.StopName;
                infoWindow.setContent(`
                    <div style="padding:10px; text-align:center;">
                        <div style="font-weight:bold; margin-bottom:5px; color:#1e40af;">${stopName}</div>
                        <i class="fas fa-spinner fa-spin" style="color:#2563eb;"></i> 查詢行經路線...
                    </div>
                `);
                activeInfoWindow = infoWindow;
                infoWindow.open(map, marker);
                fetchRealStopRoutes(stopName, infoWindow);
            });

            mapMarkers.stops.push(marker);
        });

        if (window.zoomListener) google.maps.event.removeListener(window.zoomListener);
        
        window.zoomListener = map.addListener('zoom_changed', () => {
            const currentZoom = map.getZoom();
            const threshold = 15; 
            mapMarkers.stops.forEach(marker => {
                if (currentZoom >= threshold) {
                    if (marker.content !== marker.pinContent) marker.content = marker.pinContent;
                } else {
                    if (marker.content !== marker.dotContent) marker.content = marker.dotContent;
                }
            });
        });
        google.maps.event.trigger(map, 'zoom_changed');
    }

    async function fetchRealStopRoutes(stopName, infoWindow) {
        try {
            const url = `${WORKER_URL}/?action=stop_info&city=${currentData.city}&name=${encodeURIComponent(stopName)}`;
            const res = await throttledFetch(url);
            const data = await res.json();
            const routes = data.routes || [];
            
            if (routes.length === 0) {
                infoWindow.setContent(`<div style="padding:10px; color:#999;">查無其他路線</div>`);
                return;
            }
            
            let content = `
                <div class="stop-popup">
                    <div class="stop-popup-header">
                        <i class="fas fa-map-marker-alt"></i><h3>${stopName}</h3>
                    </div>
                    <div class="stop-popup-section-title">
                        <i class="fas fa-bus"></i> 經過的路線 (${routes.length})
                    </div>
                    <div class="stop-popup-grid">
                        ${routes.map(r => `
                            <button class="stop-popup-btn" onclick="event.stopPropagation(); loadRoute('${r}', '${currentData.city}', '${currentData.type}')">${r}</button>
                        `).join('')}
                    </div>
                </div>
            `;
            infoWindow.setContent(content);

            infoWindow.setContent(`<div style="padding:10px; color:red;">查詢失敗</div>`);
        }
    }

                    infoWindow.setContent(`

                            <div style="font-size:1.1rem; font-weight:900; color:#2563eb; margin-bottom:8px;">
                                <i class="fas fa-bus"></i> ${plate}
                            </div>
                            <div style="font-size:0.9rem; color:#4b5563;">
                                <div>出廠年份：${year}</div>
                                <div style="margin-top:6px; color:#10b981; font-weight:bold;">
                                    ${isElectric ? '<i class="fas fa-bolt"></i> 電動公車' : ''}
                                    ${vReg.isLowFloor ? '<i class="fas fa-wheelchair"></i> 低底盤' : ''}
                                </div>
                            </div>
                        </div>
                    `);
                    activeInfoWindow = infoWindow;
                    infoWindow.open(map, marker);
                });
                
                mapMarkers.buses.push(marker);
            }
        });
    }

    function focusMap(lat, lng) {
        if (!map) return;
        map.panTo({ lat, lng });
        map.setZoom(17);
    }

    let userLocationMarker = null;

    function locateMe() {
        if (!navigator.geolocation) {
            alert("您的瀏覽器不支援地理定位功能。");
            return;
        }

        const btn = document.getElementById('locate-btn');
        btn.classList.add('locating');

        navigator.geolocation.getCurrentPosition(
            (position) => {
                const lat = position.coords.latitude;
                const lng = position.coords.longitude;
                btn.classList.remove('locating');

                if (!map) return;
                
                map.panTo({ lat, lng });
                map.setZoom(16);

                if (!userLocationMarker) {
                    const dot = document.createElement('div');
                    dot.style.width = '16px';
                    dot.style.height = '16px';
                    dot.style.backgroundColor = '#3b82f6';
                    dot.style.border = '3px solid white';
                    dot.style.borderRadius = '50%';
                    dot.style.boxShadow = '0 0 10px rgba(59, 130, 246, 0.5)';
                    
                    userLocationMarker = new google.maps.marker.AdvancedMarkerElement({
                        position: { lat, lng },
                        map: map,
                        content: dot,
                        title: "您的位置",
                        zIndex: 200
                    });
                } else {
                    userLocationMarker.position = { lat, lng };
                }

                if (currentData.route && currentData.stops[currentData.direction]) {
                    let minDistance = Infinity;
                    let nearestStopId = null;

                    currentData.stops[currentData.direction].forEach(s => {
                        if(s.StopPosition) {
                            const d = getDistance(lat, lng, s.StopPosition.PositionLat, s.StopPosition.PositionLon);
                            if (d < minDistance) {
                                minDistance = d;
                                nearestStopId = s.StopUID;
                            }
                        }
                    });

                    if (nearestStopId) {
                        const stopEl = document.getElementById(`stop-${nearestStopId}`);
                        if (stopEl) {
                            stopEl.scrollIntoView({ behavior: 'smooth', block: 'center' });
                            stopEl.style.backgroundColor = '#ecfdf5';
                            setTimeout(() => stopEl.style.backgroundColor = '', 2000);
                        }
                    }
                }
            },
            () => {
                btn.classList.remove('locating');
                alert("無法獲取您的位置，請確認是否開啟定位權限。");
            },
            { enableHighAccuracy: true, timeout: 5000 }
        );
    }

    async function handleSearchPlate(plate) {
        if (!plate || plate === '---') return false;
        let bus = lastBuses.find(b => (b.PlateNumb === plate || b.plate === plate));
        if (bus && bus.lat) {
            focusMap(bus.lat, bus.lon);
            return true;
        }
        alert(`找不到車牌 ${plate} 的即時位置。`);
        return false;
    }

    function decodeHtmlEntities(str) {
        if (!str) return "";
        const txt = document.createElement("textarea");
        txt.innerHTML = str;
        return txt.value;
    }

    async function openCityNewsModal(targetCity = null) {
        currentModalSession++; // 🌟 發行新的彈窗身分證
        const session = currentModalSession;

        const modal = document.getElementById('modal-overlay');
        const title = document.getElementById('modal-title');
        const body = document.getElementById('modal-body');
        
        const city = targetCity || (currentData.city ? currentData.city : 'Taipei');
        const type = (city === 'InterCity') ? 'InterCity' : 'CityBus';

        const cityNameMap = {
            'Taipei': '臺北市', 'NewTaipei': '新北市', 'Keelung': '基隆市',
            'Taoyuan': '桃園市', 'Hsinchu': '新竹市', 'HsinchuCounty': '新竹縣',
            'YilanCounty': '宜蘭縣', 'MiaoliCounty': '苗栗縣', 'Taichung': '臺中市',
            'ChanghuaCounty': '彰化縣', 'NantouCounty': '南投縣', 'YunlinCounty': '雲林縣',
            'Chiayi': '嘉義市', 'ChiayiCounty': '嘉義縣', 'Tainan': '臺南市',
            'Kaohsiung': '高雄市', 'PingtungCounty': '屏東縣',
            'HualienCounty': '花蓮縣', 'TaitungCounty': '臺東縣',
            'PenghuCounty': '澎湖縣', 'KinmenCounty': '金門縣', 'LienchiangCounty': '連江縣',
            'InterCity': '公路客運'
        };

        title.innerHTML = `<i class="fas fa-bullhorn"></i> <span>營運公告</span>`;

        let selectHtml = `
            <div style="margin-bottom: 20px; padding: 12px; background: #f1f5f9; border-radius: 12px; border: 1px solid #e2e8f0;">
                <div style="display:flex; align-items:center; gap:8px; margin-bottom:8px;">
                    <i class="fas fa-filter" style="color:#64748b;"></i>
                    <span style="font-weight:700; color:#475569; font-size:0.9rem;">切換公告縣市/類型</span>
                </div>
                <select id="news-city-select" onchange="openCityNewsModal(this.value)" style="width:100%; padding:10px 12px; border:1px solid #cbd5e1; border-radius:8px; font-size:1rem; outline:none; background:white; font-weight:600; color:#334155; cursor:pointer;">
        `;
        
        selectHtml += `<optgroup label="全台路網">`;
        const isInterSelected = (city === 'InterCity') ? 'selected' : '';
        selectHtml += `<option value="InterCity" ${isInterSelected}>公路客運</option>`;
        selectHtml += `</optgroup>`;

        selectHtml += `<optgroup label="市區公車">`;
        for (const [code, name] of Object.entries(cityNameMap)) {
            if (code === 'InterCity') continue;
            const isSelected = (code === city) ? 'selected' : '';
            selectHtml += `<option value="${code}" ${isSelected}>${name}</option>`;
        }
        selectHtml += `</optgroup></select></div>`;

        body.innerHTML = `
            ${selectHtml}
            <div id="news-content-area">
                <div style="text-align:center; padding:40px;">
                    <i class="fas fa-spinner fa-spin fa-2x" style="color:var(--primary)"></i><br><br>正在載入 ${cityNameMap[city] || city} 公告...
                </div>
            </div>
        `;
        modal.style.display = 'flex';

        try {
            const [newsRes, alertRes] = await Promise.all([
                throttledFetch(`${WORKER_URL}/?action=news&category=${type}&city=${city}`).then(r => r.json()),
                throttledFetch(`${WORKER_URL}/?action=alert&category=${type}&city=${city}`).then(r => r.json())
            ]);

            if (currentModalSession !== session) return; // 🌟 核心防護斷

            const allItems = [...(Array.isArray(alertRes) ? alertRes : []).map(a => ({ title: a.Title, desc: a.Description, date: a.StartTime, tag: '通阻', cls: 'tag-alert' })),
                              ...(Array.isArray(newsRes) ? newsRes : []).map(n => ({ title: n.Title, desc: n.Description, date: n.PublishTime, tag: '消息', cls: 'tag-news' }))]
                              .sort((a, b) => new Date(b.date) - new Date(a.date));

            const contentArea = document.getElementById('news-content-area');

            if (!allItems.length) {
                contentArea.innerHTML = `<div style="text-align:center; color:#9ca3af; padding:40px;">無公告</div>`;
                return;
            }

            contentArea.innerHTML = '<div style="display:flex; flex-direction:column; gap:12px;">' + 
                allItems.map(i => {
                    let descHtml = decodeHtmlEntities(i.desc);
                    descHtml = descHtml.replace(/<script\b[^<]*(?:(?!<\/script>)<[^<]*)*<\/script>/gi, '');
                    if (!/<[a-z][\s\S]*>/i.test(descHtml)) {
                        descHtml = descHtml.replace(/\n/g, '<br>');
                    }
                    
                    return `
                    <div style="border-bottom:1px solid #eee; padding-bottom:12px;">
                        <div style="font-size:0.8rem; color:#9ca3af; margin-bottom:4px;">${i.date ? i.date.substring(0, 10) : ''}</div>
                        <div style="font-weight:700; font-size:1rem; margin-bottom:6px;"><span class="${i.cls}">${i.tag}</span> ${i.title}</div>
                        <div class="news-desc-content">${descHtml}</div>
                    </div>`;
                }).join('') + '</div>';
        } catch (e) {
            document.getElementById('news-content-area').innerHTML = `<div style="text-align:center; color:#ef4444; padding:40px;">載入失敗</div>`;
        }
    }

    async function updateMarqueeSection(elementId, city, type) {
        const textEl = document.getElementById(elementId);
        if (!textEl) return;
        try {
            const res = await throttledFetch(`${WORKER_URL}/?action=alert&category=${type || 'CityBus'}&city=${city}`);
            const data = await res.json();
            const alerts = Array.isArray(data) ? data.filter(a => a.Status !== 1) : [];
            
            if (alerts.length > 0) {
                const text = alerts.map(a => `【${a.Title}】${a.Description}`).join("　　　");
                textEl.innerText = text;
                textEl.className = 'marquee-content';
                textEl.style.animationDuration = Math.max(10, text.length * 0.1) + 's';
            } else {
                textEl.innerText = "🚍 公車營運正常";
                textEl.className = ''; 
            }
        } catch (e) { textEl.innerText = "暫無資訊"; }
    }
